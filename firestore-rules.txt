rules_version = '2';

service cloud.firestore {
  match /databases/{database}/documents {

    match /products/{productId} {
      allow read: if true;
      allow create, update, delete: if request.auth != null;
    }

    match /customers/{customerId} {
      allow read, create, update: if request.auth != null &&
        (request.auth.uid == customerId || request.auth.token.email == "admin@sreeveerabhadra.com");
      allow delete: if request.auth != null && request.auth.token.email == "admin@sreeveerabhadra.com";
    }

    match /orders/{orderId} {
      allow create: if true;
      allow read: if request.auth != null &&
        (resource.data.userId == request.auth.uid ||
         resource.data.phone == get(/databases/$(database)/documents/customers/$(request.auth.uid)).data.phone ||
         request.auth.token.email == "admin@sreeveerabhadra.com");
      allow update, delete: if request.auth != null && request.auth.token.email == "admin@sreeveerabhadra.com";
    }

    match /coupons/{couponId} {
      allow read: if true;
      allow create, update, delete: if request.auth != null && request.auth.token.email == "admin@sreeveerabhadra.com";
    }

    match /{document=**} {
      allow read, write: if false;
    }
  }
}
